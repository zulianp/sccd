% Why the two broad phases differ, as line work: (a) a sorted order prunes on
% one axis, (b) a centroid picks one cell and the eight neighbours bound the
% search.
%
% No prose inside the drawing. What the panels mean belongs in the caption,
% where it can be read at caption size and does not compete with the marks.
\begin{tikzpicture}[
  font=\scriptsize,
  >={Stealth[length=1.6mm]},
  lbl/.style={font=\scriptsize, text=sccdInk, inner sep=1.5pt},
  ttl/.style={font=\scriptsize\bfseries, text=sccdInk, inner sep=0pt},
  done/.style={draw=sccdInk!25, fill=white, line width=0.4pt},
  test/.style={draw=sccdInk, fill=sccdInk!8, line width=0.4pt},
  prune/.style={draw=sccdInk!25, fill=white, line width=0.4pt, densely dotted},
  lead/.style={draw=sccdInk!55, line width=0.3pt},
]

% ============================ (a) sweep and prune ============================
% The walk is forward only. A box is paired with the current one when the
% current one reaches it, so the boxes above are already done; the walk stops at
% the first box whose minimum passes the current box's maximum.
\begin{scope}
  \node[ttl, anchor=south west] at (0,3.52) {(a) sweep and prune on axis $x$};

  \fill[sccdAmber!8] (0.50,0.46) rectangle (2.45,3.30);
  \draw[draw=sccdAmber!70, line width=0.4pt, densely dashed]
    (0.50,0.46) -- (0.50,3.30)  (2.45,0.46) -- (2.45,3.30);

  % Boxes sorted by their minimum on a, top to bottom.
  \foreach \i/\s/\w/\st in {
      0/0.00/1.55/done, 1/0.30/1.75/done,
      3/0.72/1.45/test, 4/0.95/2.10/test, 5/1.18/1.65/test,
      6/2.66/1.55/prune, 7/2.98/1.35/prune} {
    \pgfmathsetmacro{\y}{3.00 - 0.29*\i}
    \draw[\st] (\s,\y) rectangle ++(\w,0.17);
  }
  % The current box, whose extent is the band.
  \draw[draw=sccdAmber, fill=sccdAmber!30, line width=0.9pt]
    (0.50,2.42) rectangle ++(1.95,0.17);

  \draw[lead] (4.42,2.71) -- (4.58,2.71) -- (4.58,3.17) -- (4.42,3.17);
  \node[lbl, anchor=west, align=left] at (4.62,2.94) {prior\\steps};
  \node[lbl, anchor=west] at (4.62,2.50) {current};
  \draw[lead] (4.42,1.55) -- (4.58,1.55) -- (4.58,2.30) -- (4.42,2.30);
  \node[lbl, anchor=west, align=left] at (4.62,1.92) {tested\\against it};
  \draw[lead] (4.42,0.97) -- (4.58,0.97) -- (4.58,1.43) -- (4.42,1.43);
  \node[lbl, anchor=west] at (4.62,1.20) {pruned};

  \draw[draw=sccdInk, line width=0.4pt] (0,0.46) -- (4.30,0.46);
  \foreach \x in {0,1,2,3,4} \draw[draw=sccdInk, line width=0.4pt] (\x,0.46) -- (\x,0.38);
  \draw[->, draw=sccdInk, line width=0.4pt] (4.30,0.46) -- (4.62,0.46)
    node[lbl, anchor=west, inner sep=2pt] {$x$};
  \draw[draw=sccdAmber, line width=1.1pt] (0.50,0.20) -- (2.45,0.20);
  \node[lbl, anchor=north] at (1.48,0.14) {extent of the current box};
\end{scope}

% ============================ (b) the cell list ==============================
% The grid is drawn line by line rather than with TikZ's `grid`, which places
% its lines at multiples of the step measured from the origin and not from the
% corner of the rectangle given to it, so a cell boundary would sit where no
% boundary is and the marked cell would not line up with the box in it.
\begin{scope}[xshift=63mm]
  \node[ttl, anchor=south west] at (0,3.52) {(b) two-dimensional cell list};

  % The cells the first box covers, which are both the cells it is binned into
  % and the cells it reads.
  \fill[sccdTeal!12] (0.60,1.15) rectangle (1.80,2.95);

  \foreach \x in {0,0.60,1.20,1.80,2.40,3.00,3.60}
    \draw[draw=sccdInk!30, line width=0.3pt] (\x,0.55) -- (\x,3.55);
  \foreach \y in {0.55,1.15,1.75,2.35,2.95,3.55}
    \draw[draw=sccdInk!30, line width=0.3pt] (0,\y) -- (3.60,\y);
  \draw[draw=sccdInk!45, line width=0.4pt] (0,0.55) rectangle (3.60,3.55);

  % The cells the second box covers, outlined rather than filled so the two
  % ranges can overlap and still be read apart.
  \draw[draw=sccdPlum!55, line width=0.5pt, densely dashed] (1.20,1.75) rectangle (3.00,2.95);

  \fill[sccdAmber!45] (1.40,1.90) rectangle (1.75,2.35);
  \draw[draw=sccdInk, line width=1.0pt] (1.20,1.75) rectangle (1.80,2.35);

  \draw[draw=sccdTeal, line width=0.9pt] (0.70,1.30) rectangle (1.75,2.35);
  \draw[draw=sccdPlum, line width=0.9pt] (1.40,1.90) rectangle (2.45,2.90);
  \fill[sccdInk] (1.40,1.90) circle (1.6pt);

  \node[lbl, text=sccdTeal, anchor=east] at (0.66,1.42) {first};
  \node[lbl, text=sccdPlum, anchor=west] at (2.49,2.80) {second};

  % --- key in one column, so no label can run into the next ---
  \fill[sccdTeal!12] (0.00,0.24) rectangle ++(0.22,0.16);
  \draw[draw=sccdInk!30, line width=0.3pt] (0.00,0.24) rectangle ++(0.22,0.16);
  \node[lbl, anchor=west] at (0.26,0.32) {the cells the first box covers and reads};
  \fill[sccdAmber!45] (0.00,0.02) rectangle ++(0.22,0.16);
  \node[lbl, anchor=west] at (0.26,0.10) {overlap of the two boxes};
  \draw[draw=sccdInk, line width=1.0pt] (0.00,-0.20) rectangle ++(0.22,0.16);
  \node[lbl, anchor=west] at (0.26,-0.12) {the cell that emits the pair};
\end{scope}
\end{tikzpicture}
