% Why the two broad phases differ. Left: sorting on one axis prunes on one axis,
% which a draped surface defeats. Right: the cell list bins instead, and settles
% the duplicates that binning creates with a rule that carries no per-pair state.
%
% The window is drawn as a band through the whole sorted list rather than as an
% arrow beneath it, because the point of the left panel is which boxes the band
% crosses. Boxes are shaded by what happens to them -- pruned, tested and
% rejected, tested and found -- so the cost the prose quantifies is visible.
\begin{tikzpicture}[
  font=\footnotesize,
  >={Stealth[length=2mm]},
  pruned/.style={draw=sccdGrid!60!sccdInk, fill=white, line width=0.35pt},
  tested/.style={draw=sccdAmber, fill=sccdAmber!16, line width=0.35pt},
  found/.style={draw=sccdTeal, fill=sccdTeal!22, line width=0.8pt},
  hd/.style={font=\footnotesize\bfseries, text=sccdInk, inner sep=2pt},
  tag/.style={font=\scriptsize, inner sep=2pt},
  note/.style={font=\scriptsize, text=sccdInk, align=left, text width=63mm,
               anchor=north west, inner sep=0pt},
]

% ======================= left: sweep and prune ==========================
\begin{scope}
  \node[hd, anchor=south west] at (0,4.46) {(a)\;\;sweep and prune along axis $a$};

  % The query's extent on a, as a band behind the list. Everything it crosses is
  % a candidate; everything it misses is pruned by the sort alone.
  \fill[sccdAmber!10] (0.60,1.16) rectangle (2.60,4.34);
  \draw[draw=sccdAmber, line width=0.5pt, densely dashed]
    (0.60,1.16) -- (0.60,4.34)  (2.60,1.16) -- (2.60,4.34);

  % Boxes sorted by their minimum on a, one per row. Six meet the window; of
  % those, one is a real collision and five fail on another axis.
  \foreach \i/\s/\w/\st in {
      0/0.00/1.70/tested, 1/0.35/1.90/tested, 2/0.55/2.10/found,
      3/0.80/1.60/tested, 4/1.05/2.30/tested, 5/1.30/1.80/tested,
      6/2.95/1.70/pruned, 7/3.30/1.50/pruned} {
    \pgfmathsetmacro{\y}{4.06 - 0.36*\i}
    \draw[\st] (\s,\y) rectangle ++(\w,0.22);
  }

  % What happened to each group, named beside the rows rather than above them.
  \draw[draw=sccdAmber, line width=0.6pt] (4.74,2.26) -- (4.86,2.26)
    -- (4.86,4.28) -- (4.74,4.28);
  \node[tag, text=sccdAmber, anchor=west, align=left] at (4.90,3.27)
    {tested\\in full};
  \draw[draw=sccdGrid!60!sccdInk, line width=0.6pt] (4.74,1.54) -- (4.86,1.54)
    -- (4.86,2.12) -- (4.74,2.12);
  \node[tag, text=sccdGrid!50!sccdInk, anchor=west] at (4.90,1.83) {pruned};

  \node[tag, text=sccdTeal, anchor=west] at (2.70,3.45) {a real overlap};
  \draw[->, draw=sccdTeal, line width=0.5pt] (2.68,3.45) -- (2.36,3.45);

  \draw[->, draw=sccdInk] (-0.05,0.70) -- (5.05,0.70)
    node[anchor=west, inner sep=3pt] {$a$};
  \node[tag, text=sccdAmber, anchor=north] at (1.60,1.12)
    {the query's extent on $a$};

  \node[note] at (-0.05,0.42)
    {Every box the band crosses overlaps the query on $a$ and is therefore
     tested in full. On a draped surface almost all of them then fail on one of
     the other two axes: about $2100$ candidates are walked for each pair
     actually found, and no choice of axis rescues it. The two good axes give
     $2882$ and $2838$ candidates per pair, the third $126\,063$.};
\end{scope}

% ========================= right: 2D cell list ==========================
\begin{scope}[xshift=74mm]

  \fill[sccdGrid!18] (0,1.70) rectangle (4.96,4.18);

  \draw[draw=sccdGrid!70!sccdInk, line width=0.5pt] (0,1.70) rectangle (4.96,4.18);

  \draw[draw=sccdTeal, fill=sccdTeal!12, line width=0.9pt]

  \fill[sccdAmber!60] (2.10,3.10) rectangle (2.80,3.55);
  \draw[draw=sccdInk, line width=1.2pt, densely dashed]

  \fill[sccdInk] (2.10,3.10) circle (2.1pt);
  \draw[draw=sccdInk, line width=0.5pt] (2.10,3.10) -- (1.62,2.62);
  \node[tag, text=sccdInk, anchor=north east] at (1.64,2.64) {minimum corner};

  % --- legend, one item per line so nothing collides at this size ---
  \fill[sccdAmber!60] (0.05,1.24) rectangle ++(0.28,0.20);
  \node[font=\scriptsize, anchor=west, inner sep=3pt] at (0.33,1.34)
    {the overlap of the two boxes};
  \draw[draw=sccdInk, line width=1.2pt, densely dashed]
    (0.05,0.92) rectangle ++(0.28,0.20);
  \node[font=\scriptsize, anchor=west, inner sep=3pt] at (0.33,1.02)
    {the cell that owns the pair};

  \node[note] at (-0.05,0.42)
    {Each box is binned into every cell it touches, so the same pair can be met
     in several of them. It is emitted only by the cell holding the minimum
     corner of the overlap. Since that corner lies inside both boxes, both are
     binned there and the cell is unique: two clamps per surviving pair, and no
     per-pair state to allocate.};
\end{scope}
\end{tikzpicture}
